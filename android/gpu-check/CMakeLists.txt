cmake_minimum_required(VERSION 3.24)
project(tempest_android_gpu_check LANGUAGES C CXX)
if(NOT ANDROID OR NOT ANDROID_ABI STREQUAL "arm64-v8a")
    message(FATAL_ERROR "Use the Android NDK toolchain with arm64-v8a")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(TEMPEST_ADRENOTOOLS_SOURCE "" CACHE PATH "Pinned libadrenotools checkout")
if(NOT EXISTS "${TEMPEST_ADRENOTOOLS_SOURCE}/CMakeLists.txt")
    message(FATAL_ERROR "Set TEMPEST_ADRENOTOOLS_SOURCE")
endif()
set(BUILD_SHARED_LIBS OFF)
add_subdirectory("${TEMPEST_ADRENOTOOLS_SOURCE}" adrenotools EXCLUDE_FROM_ALL)
add_library(tempest-gpu-check SHARED gpu_check.cpp)
target_compile_options(tempest-gpu-check PRIVATE -Wall -Wextra -Werror
    -Wno-missing-field-initializers -Wno-missing-designated-field-initializers)
target_link_libraries(tempest-gpu-check PRIVATE adrenotools android log dl)
# These hooks are loaded by name by libadrenotools.
add_dependencies(tempest-gpu-check main_hook hook_impl file_redirect_hook gsl_alloc_hook)
